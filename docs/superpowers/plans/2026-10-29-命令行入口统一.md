> **For agentic workers:** Read `subagent-driven-development.md` when the user authorizes subagents; otherwise read `executing-plans.md` and implement task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Tech Stack:** Windows、Python 3.14、pandas、openpyxl、pytest。

**Spec:** `docs/superpowers/specs/2026-10-29-命令行入口统一-design.md`

```python
# -*- coding: utf-8 -*-
import subprocess
import sys
from pathlib import Path

import pytest

ROOT = Path(__file__).resolve().parents[1]
MAIN = ROOT / "main.py"
CLI = ROOT / "cli" / "序时账分析器命令行版.py"


def run_entry(entry, *args):
    return subprocess.run(
        [sys.executable, "-B", "-X", "utf8", str(entry), *map(str, args)],
        cwd=ROOT, capture_output=True, text=True, encoding="utf-8",
        errors="replace", timeout=90,
    )


@pytest.mark.parametrize("entry", [MAIN, CLI])
def test_无头入口显示当前版本(entry):
    finished = run_entry(entry, "--help")
    assert finished.returncode == 0
    assert "v2.5.0" in finished.stdout
```
